# 脚本管理模块需求

## 1. 模块概述

脚本管理模块负责统一管理所有运维脚本资源，将原有分散的CGI脚本整合为可维护、可复用的脚本库。提供脚本编辑、版本控制、参数化配置等功能。

### 3.4 版本控制系统

#### 3.4.1 版本管理
```javascript
class ScriptVersionControl {
  // 版本号规则：major.minor.patch
  incrementVersion(currentVersion, type = 'patch') {
    const [major, minor, patch] = currentVersion.split('.').map(Number);
    
    switch(type) {
      case 'major':
        return `${major + 1}.0.0`;
      case 'minor':
        return `${major}.${minor + 1}.0`;
      case 'patch':
      default:
        return `${major}.${minor}.${patch + 1}`;
    }
  }
  
  // 创建版本快照
  async createSnapshot(script) {
    const snapshot = {
      scriptId: script.id,
      version: script.version,
      content: script.content,
      parameters: script.parameters,
      changelog: script.changelog || '',
      createdAt: new Date(),
      createdBy: getCurrentUser()
    };
    
    return await this.saveSnapshot(snapshot);
  }
  
  // 版本对比
  async compareVersions(scriptId, version1, version2) {
    const v1 = await this.getVersion(scriptId, version1);
    const v2 = await this.getVersion(scriptId, version2);
    
    return this.generateDiff(v1.content, v2.content);
  }
  
  // 版本回滚
  async rollbackToVersion(scriptId, targetVersion) {
    const targetScript = await this.getVersion(scriptId, targetVersion);
    const currentScript = await this.getCurrentScript(scriptId);
    
    // 创建当前版本快照
    await this.createSnapshot(currentScript);
    
    // 回滚到目标版本
    return await this.updateScript(scriptId, {
      content: targetScript.content,
      parameters: targetScript.parameters,
      version: this.incrementVersion(currentScript.version, 'minor'),
      changelog: `回滚到版本 ${targetVersion}`
    });
  }
}
```

// 文档生成器
class ScriptDocumentationGenerator {
  generateDoc(script) {
    const doc = {
      basicInfo: this.extractBasicInfo(script),
      parameters: this.extractParameters(script),
      examples: this.generateExamples(script),
      changelog: this.getChangelog(script.id),
      dependencies: this.analyzeDependencies(script),
      complexity: this.analyzeComplexity(script)
    };
    
    return this.renderDocumentation(doc);
  }
  
  // 分析脚本复杂度
  analyzeComplexity(script) {
    const lines = script.content.split('\n').length;
    const functions = (script.content.match(/function\s+\w+/g) || []).length;
    const loops = (script.content.match(/for|while/g) || []).length;
    
    return {
      lines,
      functions,
      loops,
      complexity: this.calculateComplexity(lines, functions, loops)
    };
  }
  
  // 依赖分析
  analyzeDependencies(script) {
    const commands = this.extractCommands(script.content);
    const dependencies = commands.map(cmd => ({
      command: cmd,
      required: true,
      available: this.checkCommandAvailability(cmd)
    }));
    
    return dependencies;
  }
}

### 6.1 数据库设计
```sql
-- 脚本主表
CREATE TABLE scripts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    category TEXT NOT NULL,
    description TEXT,
    content TEXT NOT NULL,
    version TEXT NOT NULL DEFAULT '1.0.0',
    author TEXT,
    tags TEXT,                    -- JSON格式存储标签
    parameters TEXT,              -- JSON格式存储参数配置
    is_active BOOLEAN DEFAULT 1,  -- 是否激活
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 脚本版本历史表
CREATE TABLE script_versions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    script_id INTEGER NOT NULL,
    version TEXT NOT NULL,
    content TEXT NOT NULL,
    parameters TEXT,
    changelog TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    created_by TEXT,
    FOREIGN KEY(script_id) REFERENCES scripts(id) ON DELETE CASCADE
);

-- 脚本分类表
CREATE TABLE script_categories (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL UNIQUE,
    parent_id INTEGER,
    description TEXT,
    sort_order INTEGER DEFAULT 0,
    FOREIGN KEY(parent_id) REFERENCES script_categories(id)
);

-- 创建索引
CREATE INDEX idx_scripts_category ON scripts(category);
CREATE INDEX idx_scripts_name ON scripts(name);
CREATE INDEX idx_script_versions_script_id ON script_versions(script_id);
```

### 6.2 脚本执行沙箱
```javascript
// 脚本安全执行环境
class ScriptSandbox {
  constructor() {
    this.allowedCommands = [
      'ls', 'cat', 'grep', 'awk', 'sed', 'sort', 'uniq',
      'ps', 'df', 'free', 'uname', 'whoami', 'id'
    ];
    
    this.blockedCommands = [
      'rm', 'rmdir', 'dd', 'mkfs', 'fdisk',
      'shutdown', 'reboot', 'halt', 'init'
    ];
  }
  
  // 脚本安全检查
  validateScript(scriptContent) {
    const commands = this.extractCommands(scriptContent);
    const dangerous = commands.filter(cmd => 
      this.blockedCommands.includes(cmd)
    );
    
    if (dangerous.length > 0) {
      throw new Error(`检测到危险命令: ${dangerous.join(', ')}`);
    }
    
    return true;
  }
  
  // 执行环境隔离
  async executeInSandbox(script, parameters, hosts) {
    // 1. 参数替换
    const processedScript = this.replaceParameters(script, parameters);
    
    // 2. 安全检查
    this.validateScript(processedScript);
    
    // 3. 创建临时执行文件
    const tempScript = await this.createTempScript(processedScript);
    
    // 4. 执行脚本
    const results = await this.executeOnHosts(tempScript, hosts);
    
    // 5. 清理临时文件
    await this.cleanup(tempScript);
    
    return results;
  }
}
```
