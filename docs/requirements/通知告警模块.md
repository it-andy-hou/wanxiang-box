# 通知告警模块需求

## 1. 模块概述

通知告警模块是万象匣的重要辅助功能，提供企业微信集成和多场景智能通知能力。通过实时推送关键事件信息，帮助运维人员及时了解系统状态变化，提升运维响应效率。

```javascript
// 企业微信机器人配置
class WeChatNotifier {
  constructor(webhookUrl, secret) {
    this.webhookUrl = webhookUrl;
    this.secret = secret;
    this.rateLimiter = new RateLimiter(20, 60000); // 每分钟最多20条消息
  }
  
  async sendMessage(message, options = {}) {
    try {
      // 构建消息内容
      const payload = this.buildMessage(message, options);
      
      // 签名验证
      const signature = this.generateSignature(payload);
      
      // 发送消息
      const response = await this.sendToWeChat(payload, signature);
      
      return { success: true, messageId: response.msgid };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }
  
  buildMessage(message, options) {
    const { type = 'text', mentions = [] } = options;
    
    switch (type) {
      case 'text':
        return {
          msgtype: 'text',
          text: {
            content: message,
            mentioned_list: mentions.map(m => `@${m}`)
          }
        };
        
      case 'markdown':
        return {
          msgtype: 'markdown',
          markdown: {
            content: message
          }
        };
        
      case 'card':
        return {
          msgtype: 'template_card',
          template_card: message
        };
    }
  }
}
```

### 2.2 通知场景管理

#### 2.2.1 主机状态变更通知
```javascript
class HostNotificationManager {
  async onHostAdded(host, operator) {
    const message = this.renderTemplate('hostAdded', {
      hostInfo: `${host.ip} (${host.systemName})`,
      timestamp: new Date().toLocaleString(),
      operator: operator.name,
      status: host.status === 'online' ? '🟢 在线' : '🔴 离线'
    });
    
    await this.sendNotification(message, {
      type: 'markdown',
      mentions: [operator.name],
      priority: 'normal'
    });
  }
  
  async onHostStatusChanged(host, oldStatus, newStatus) {
    if (this.shouldNotifyStatusChange(oldStatus, newStatus)) {
      const message = `
### 📡 主机状态变更
**主机**：${host.ip} (${host.systemName})
**状态变更**：${this.getStatusIcon(oldStatus)} ${oldStatus} → ${this.getStatusIcon(newStatus)} ${newStatus}
**变更时间**：${new Date().toLocaleString()}
      `;
      
      await this.sendNotification(message, {
        type: 'markdown',
        priority: newStatus === 'offline' ? 'high' : 'normal'
      });
    }
  }
  
  shouldNotifyStatusChange(oldStatus, newStatus) {
    // 只有关键状态变更才通知
    const criticalChanges = [
      'online -> offline',
      'offline -> online',
      'unknown -> offline'
    ];
    
    return criticalChanges.includes(`${oldStatus} -> ${newStatus}`);
  }
}
```

#### 2.2.2 任务执行结果通知
```javascript
class TaskNotificationManager {
  async onTaskCompleted(task, results) {
    const successCount = results.filter(r => r.success).length;
    const totalCount = results.length;
    const successRate = Math.round((successCount / totalCount) * 100);
    
    const message = this.renderTemplate('taskCompleted', {
      taskName: task.name,
      successRate: successRate,
      hostCount: totalCount,
      duration: this.formatDuration(task.duration),
      operator: task.operator
    });
    
    await this.sendNotification(message, {
      type: 'card',
      mentions: [task.operator],
      priority: successRate < 80 ? 'high' : 'normal'
    });
    
    // 如果有失败的主机，发送详细信息
    if (successRate < 100) {
      await this.sendFailureDetails(task, results);
    }
  }
  
  async sendFailureDetails(task, results) {
    const failures = results.filter(r => !r.success);
    
    const message = `
### ❌ 任务执行失败详情
**任务名称**：${task.name}
**失败主机数量**：${failures.length}

${failures.map(f => `- ${f.host.ip}: ${f.error}`).join('\n')}

> 请检查相关主机状态和网络连接
    `;
    
    await this.sendNotification(message, {
      type: 'markdown',
      priority: 'high'
    });
  }
}
```

### 2.3 告警规则配置

#### 2.3.1 告警规则管理
```javascript
class AlertRuleManager {
  constructor() {
    this.rules = new Map();
    this.loadDefaultRules();
  }
  
  loadDefaultRules() {
    // CPU使用率告警
    this.addRule({
      id: 'cpu_high',
      name: 'CPU使用率过高',
      condition: 'cpu_usage > 80',
      severity: 'warning',
      duration: 300, // 持续5分钟
      cooldown: 1800, // 冷却30分钟
      enabled: true
    });
    
    // 内存使用率告警
    this.addRule({
      id: 'memory_high',
      name: '内存使用率过高',
      condition: 'memory_usage > 85',
      severity: 'warning',
      duration: 300,
      cooldown: 1800,
      enabled: true
    });
    
    // 磁盘空间告警
    this.addRule({
      id: 'disk_full',
      name: '磁盘空间不足',
      condition: 'disk_usage > 90',
      severity: 'critical',
      duration: 60,
      cooldown: 3600,
      enabled: true
    });
    
    // SSH连接失败告警
    this.addRule({
      id: 'ssh_failed',
      name: 'SSH连接失败',
      condition: 'ssh_status == "failed"',
      severity: 'critical',
      duration: 0,
      cooldown: 300,
      enabled: true
    });
  }
  
  evaluateRules(hostMetrics) {
    const alerts = [];
    
    for (const [ruleId, rule] of this.rules) {
      if (!rule.enabled) continue;
      
      const result = this.evaluateCondition(rule.condition, hostMetrics);
      
      if (result && !this.isInCooldown(ruleId, hostMetrics.hostId)) {
        alerts.push({
          ruleId,
          rule,
          hostId: hostMetrics.hostId,
          timestamp: Date.now(),
          value: hostMetrics
        });
      }
    }
    
    return alerts;
  }
}
```

### 2.4 通知优先级和频率控制

#### 2.4.1 优先级管理
```javascript
class NotificationPriorityManager {
  constructor() {
    this.priorities = {
      low: { delay: 0, rateLimit: 10 },      // 普通信息
      normal: { delay: 0, rateLimit: 20 },   // 一般通知
      high: { delay: 0, rateLimit: 50 },     // 重要告警
      critical: { delay: 0, rateLimit: 100 } // 紧急告警
    };
    
    this.queues = {
      low: [],
      normal: [],
      high: [],
      critical: []
    };
  }
  
  async enqueueNotification(message, priority = 'normal') {
    const notification = {
      id: this.generateId(),
      message,
      priority,
      timestamp: Date.now(),
      retryCount: 0
    };
    
    this.queues[priority].push(notification);
    this.processQueue(priority);
  }
  
  async processQueue(priority) {
    const queue = this.queues[priority];
    const config = this.priorities[priority];
    
    while (queue.length > 0) {
      const notification = queue.shift();
      
      try {
        await this.sendNotification(notification);
        this.recordSuccess(notification);
      } catch (error) {
        await this.handleFailure(notification, error);
      }
      
      // 速率限制
      if (queue.length > 0) {
        await this.delay(config.rateLimit);
      }
    }
  }
}
```

### 3.1 数据库设计
```sql
-- 通知配置表
CREATE TABLE notification_configs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    type TEXT NOT NULL,              -- 'wechat' | 'email' | 'sms'
    config TEXT NOT NULL,            -- JSON配置
    enabled BOOLEAN DEFAULT true,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 告警规则表
CREATE TABLE alert_rules (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    condition_expr TEXT NOT NULL,    -- 告警条件表达式
    severity TEXT NOT NULL,          -- 'info' | 'warning' | 'critical'
    duration INTEGER DEFAULT 0,      -- 持续时间（秒）
    cooldown INTEGER DEFAULT 300,    -- 冷却时间（秒）
    enabled BOOLEAN DEFAULT true,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 通知历史表
CREATE TABLE notification_history (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    type TEXT NOT NULL,
    recipient TEXT NOT NULL,
    subject TEXT,
    content TEXT NOT NULL,
    priority TEXT DEFAULT 'normal',
    status TEXT DEFAULT 'pending',   -- 'pending' | 'sent' | 'failed'
    sent_at DATETIME,
    error_message TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 告警历史表
CREATE TABLE alert_history (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    rule_id INTEGER,
    host_id INTEGER,
    alert_level TEXT NOT NULL,
    message TEXT NOT NULL,
    resolved BOOLEAN DEFAULT false,
    resolved_at DATETIME,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(rule_id) REFERENCES alert_rules(id),
    FOREIGN KEY(host_id) REFERENCES hosts(id)
);
```

```javascript
// 企业微信机器人API封装
class WeChatBotAPI {
  constructor(webhookUrl) {
    this.webhookUrl = webhookUrl;
    this.axios = axios.create({
      timeout: 10000,
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  async sendText(content, mentionedList = []) {
    const payload = {
      msgtype: 'text',
      text: {
        content,
        mentioned_list: mentionedList
      }
    };
    
    return await this.send(payload);
  }
  
  async sendMarkdown(content) {
    const payload = {
      msgtype: 'markdown',
      markdown: { content }
    };
    
    return await this.send(payload);
  }
  
  async send(payload) {
    try {
      const response = await this.axios.post(this.webhookUrl, payload);
      
      if (response.data.errcode === 0) {
        return { success: true, data: response.data };
      } else {
        throw new Error(response.data.errmsg || '发送失败');
      }
    } catch (error) {
      return { success: false, error: error.message };
    }
  }
}
```
