## 2. 数据备份与恢复

### 2.1 自动备份功能

#### 2.1.1 配置数据备份
```javascript
class ConfigBackupManager {
  constructor() {
    this.backupSchedule = '0 2 * * *'; // 每天凌晨2点自动备份
    this.maxBackupFiles = 30;          // 保留30天备份
    this.backupPath = './backups/configs';
  }
  
  async createBackup(type = 'auto') {
    const timestamp = new Date().toISOString().replace(/:/g, '-');
    const backupData = {
      metadata: {
        version: '1.0',
        timestamp,
        type, // 'auto' | 'manual'
        creator: this.getCurrentUser()
      },
      data: {
        hosts: await this.exportHosts(),
        scripts: await this.exportScripts(),
        users: await this.exportUsers(),
        settings: await this.exportSettings(),
        alertRules: await this.exportAlertRules()
      }
    };
    
    const filename = `config_backup_${timestamp}.json`;
    const filepath = path.join(this.backupPath, filename);
    
    // 压缩并加密备份文件
    const compressed = await this.compressData(backupData);
    const encrypted = await this.encryptData(compressed);
    
    await fs.writeFile(filepath, encrypted);
    
    // 清理旧备份
    await this.cleanupOldBackups();
    
    return {
      success: true,
      filename,
      size: encrypted.length,
      timestamp
    };
  }
}
```

## 3. 日志管理与审计

### 3.1 操作日志记录

#### 3.1.1 全操作审计
```javascript
class AuditLogger {
  async logOperation(operation) {
    const logEntry = {
      user_id: operation.userId,
      action_type: operation.action,
      resource_type: operation.resourceType,
      resource_id: operation.resourceId,
      details: JSON.stringify(operation.details),
      ip_address: operation.ipAddress,
      user_agent: operation.userAgent,
      result: operation.result,
      timestamp: new Date().toISOString(),
      session_id: operation.sessionId
    };
    
    await this.db.query(`
      INSERT INTO audit_logs 
      (user_id, action_type, resource_type, resource_id, details, ip_address, user_agent, result, timestamp, session_id)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `, Object.values(logEntry));
    
    // 异步处理敏感操作告警
    if (this.isSensitiveOperation(operation)) {
      this.processSensitiveOperationAlert(logEntry);
    }
  }
  
  isSensitiveOperation(operation) {
    const sensitiveActions = [
      'host_delete',
      'script_delete', 
      'user_delete',
      'config_change',
      'backup_restore'
    ];
    
    return sensitiveActions.includes(operation.action);
  }
}
```

```javascript
class LogSearchEngine {
  async searchLogs(searchQuery, options = {}) {
    const {
      logType = 'all',
      dateRange = {},
      limit = 100,
      offset = 0
    } = options;
    
    let results = [];
    
    // 操作日志搜索
    if (logType === 'all' || logType === 'audit') {
      const auditResults = await this.searchAuditLogs(searchQuery, dateRange, limit);
      results.push(...auditResults.map(r => ({ ...r, type: 'audit' })));
    }
    
    // 执行日志搜索
    if (logType === 'all' || logType === 'execution') {
      const executionResults = await this.searchExecutionLogs(searchQuery, dateRange, limit);
      results.push(...executionResults.map(r => ({ ...r, type: 'execution' })));
    }
    
    // 按时间排序
    results.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
    
    return {
      results: results.slice(offset, offset + limit),
      total: results.length,
      query: searchQuery,
      searchTime: Date.now()
    };
  }
}
```

### 4.1 数据库扩展设计
```sql
-- 配置版本表
CREATE TABLE config_versions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    table_name TEXT NOT NULL,
    record_id INTEGER NOT NULL,
    user_id INTEGER,
    changes TEXT NOT NULL,
    timestamp DATETIME NOT NULL,
    version_number INTEGER NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 备份记录表  
CREATE TABLE backup_records (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    filename TEXT NOT NULL,
    type TEXT NOT NULL,           -- 'full' | 'incremental' | 'manual'
    size INTEGER NOT NULL,
    checksum TEXT NOT NULL,
    created_by INTEGER,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(created_by) REFERENCES users(id)
);

-- 脚本执行日志表
CREATE TABLE script_execution_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    task_id TEXT NOT NULL,
    script_id INTEGER,
    host_id INTEGER,
    user_id INTEGER,
    start_time DATETIME NOT NULL,
    end_time DATETIME,
    exit_code INTEGER,
    stdout TEXT,
    stderr TEXT,
    command TEXT NOT NULL,
    parameters TEXT,
    status TEXT NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(script_id) REFERENCES scripts(id),
    FOREIGN KEY(host_id) REFERENCES hosts(id),
    FOREIGN KEY(user_id) REFERENCES users(id)
);

-- 系统日志表
CREATE TABLE system_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    level TEXT NOT NULL,          -- 'info' | 'warning' | 'error' | 'critical'
    component TEXT NOT NULL,      -- 'auth' | 'ssh' | 'backup' | 'notification'
    message TEXT NOT NULL,
    details TEXT,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
);
```
