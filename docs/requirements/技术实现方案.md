class HostManager {
  constructor() {
    this.db = new Database('hosts.db');
    this.sshManager = new SSHManager();
  }
  
  // 添加主机并自动配置
  async addHost(hostInfo) {
    // 1. 验证主机信息

    await this.testConnection(hostInfo);
    
    // 3. 配置SSH免密
    await this.setupSSHKey(hostInfo);

    // 5. 保存到数据库

  }
}

### 2.2 脚本管理器
```javascript
class ScriptManager {
  constructor() {
    this.db = new Database('scripts.db');
    this.editor = new MonacoEditor();
  }
  
  // 脚本参数化替换
  processScript(script, parameters) {
    let processedContent = script.content;
    
    for (const [key, value] of Object.entries(parameters)) {
      const placeholder = `{{${key.toUpperCase()}}}`;
      processedContent = processedContent.replace(
        new RegExp(placeholder, 'g'), 
        value
      );
    }
    
    return processedContent;
  }
}
```

### 2.3 任务执行器
```javascript
class TaskExecutor {
  constructor(maxConcurrency = 20) {
    this.maxConcurrency = maxConcurrency;
    this.activeConnections = new Map();
    this.taskQueue = [];
  }
  
  async executeBatch(script, hosts, parameters) {
    const results = [];
    const semaphore = new Semaphore(this.maxConcurrency);
    
    const promises = hosts.map(async (host) => {
      await semaphore.acquire();
      
      try {
        const result = await this.executeOnHost(script, host, parameters);
        results.push(result);
      } finally {
        semaphore.release();
      }
    });
    
    await Promise.all(promises);
    return results;
  }
}
```

### 2.4 文件传输管理器
```javascript
class FileTransferManager {
  constructor() {
    this.db = new Database('transfers.db');
    this.sftpPool = new SFTPConnectionPool();
    this.transferQueue = new TransferQueue();
  }
  
  // 文件上传
  async uploadFiles(files, targetHost, remotePath, options = {}) {
    const results = [];
    
    for (const file of files) {
      const transferTask = {
        type: 'upload',
        localPath: file.path,
        remotePath: path.join(remotePath, file.name),
        hostInfo: targetHost,
        options
      };
      
      const taskId = this.transferQueue.addTransfer(transferTask);
      results.push({ file: file.name, taskId });
    }
    
    return results;
  }
  
  // 批量文件下载
  async downloadFiles(files, sourceHost, localPath, options = {}) {
    const results = [];
    
    for (const file of files) {
      const transferTask = {
        type: 'download',
        remotePath: file.path,
        localPath: path.join(localPath, file.name),
        hostInfo: sourceHost,
        options
      };
      
      const taskId = this.transferQueue.addTransfer(transferTask);
      results.push({ file: file.name, taskId });
    }
    
    return results;
  }
  
  // 目录同步
  async syncDirectories(localDir, remoteDir, hostInfo, strategy = 'bidirectional') {
    const localFiles = await this.scanLocalDirectory(localDir);
    const remoteFiles = await this.scanRemoteDirectory(remoteDir, hostInfo);
    
    const syncPlan = this.generateSyncPlan(localFiles, remoteFiles, strategy);
    return await this.executeSyncPlan(syncPlan, hostInfo);
  }
}
```

### 2.5 通知告警管理器
```javascript
class NotificationManager {
  constructor() {
    this.wechatBot = new WeChatBotAPI();
    this.alertRules = new AlertRuleManager();
    this.priorityQueue = new NotificationPriorityManager();
  }
  
  // 企业微信通知
  async sendWeChatNotification(message, options = {}) {
    const { type = 'text', mentions = [], priority = 'normal' } = options;
    
    await this.priorityQueue.enqueueNotification({
      type: 'wechat',
      message,
      options: { type, mentions },
      priority
    });
  }
  
  // 主机状态变更通知
  async notifyHostStatusChange(host, oldStatus, newStatus) {
    if (this.shouldNotifyStatusChange(oldStatus, newStatus)) {
      const message = this.buildHostStatusMessage(host, oldStatus, newStatus);
      await this.sendWeChatNotification(message, { priority: 'high' });
    }
  }
  
  // 任务执行结果通知
  async notifyTaskCompletion(task, results) {
    const successRate = this.calculateSuccessRate(results);
    const message = this.buildTaskCompletionMessage(task, successRate);
    
    await this.sendWeChatNotification(message, {
      type: 'markdown',
      priority: successRate < 80 ? 'high' : 'normal'
    });
  }
}
```

### 2.6 数据管理器
```javascript
class DataManager {
  constructor() {
    this.backupManager = new ConfigBackupManager();
    this.auditLogger = new AuditLogger();
    this.logRotation = new LogRotationManager();
  }
  
  // 数据备份
  async createBackup(type = 'auto', options = {}) {
    try {
      const result = await this.backupManager.createBackup(type);
      
      await this.auditLogger.logOperation({
        action: 'backup_created',
        resourceType: 'backup',
        resourceId: result.filename,
        result: 'success',
        details: { size: result.size, type }
      });
      
      return result;
    } catch (error) {
      await this.auditLogger.logOperation({
        action: 'backup_failed',
        resourceType: 'backup',
        result: 'failed',
        details: { error: error.message, type }
      });
      
      throw error;
    }
  }
  
  // 日志查询
  async queryLogs(filters) {
    const logSearch = new LogSearchEngine();
    return await logSearch.searchLogs(filters.query, {
      logType: filters.type,
      dateRange: filters.dateRange,
      limit: filters.limit,
      offset: filters.offset
    });
  }
  
  // 定期维护
  async performMaintenance() {
    // 日志轮转
    await this.logRotation.performLogRotation();
    
    // 数据库优化
    await this.optimizeDatabase();
    
    // 自动备份
    await this.createBackup('scheduled');
  }
}

## 3. 数据库设计

### 3.1 数据库表结构
```sql
-- 主机表
CREATE TABLE hosts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    system_name TEXT NOT NULL,
    app_name TEXT NOT NULL,
    ip TEXT NOT NULL UNIQUE,
    username TEXT DEFAULT 'root',
    datacenter TEXT,
    status TEXT DEFAULT 'unknown',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 脚本表
CREATE TABLE scripts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    category TEXT NOT NULL,
    content TEXT NOT NULL,
    version TEXT DEFAULT '1.0.0',
    parameters TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 任务表
CREATE TABLE tasks (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    script_id INTEGER,
    status TEXT DEFAULT 'pending',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(script_id) REFERENCES scripts(id)
);

-- 文件传输记录表
CREATE TABLE file_transfers (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    type TEXT NOT NULL,
    host_id INTEGER,
    local_path TEXT NOT NULL,
    remote_path TEXT NOT NULL,
    file_size INTEGER DEFAULT 0,
    transferred_size INTEGER DEFAULT 0,
    status TEXT DEFAULT 'pending',
    start_time DATETIME,
    end_time DATETIME,
    error_message TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(host_id) REFERENCES hosts(id)
);

-- 通知配置表
CREATE TABLE notification_configs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    type TEXT NOT NULL,              -- 'wechat' | 'email' | 'sms'
    config TEXT NOT NULL,            -- JSON配置
    enabled BOOLEAN DEFAULT true,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 告警规则表
CREATE TABLE alert_rules (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    condition_expr TEXT NOT NULL,    -- 告警条件表达式
    severity TEXT NOT NULL,          -- 'info' | 'warning' | 'critical'
    duration INTEGER DEFAULT 0,      -- 持续时间（秒）
    cooldown INTEGER DEFAULT 300,    -- 冷却时间（秒）
    enabled BOOLEAN DEFAULT true,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 审计日志表
CREATE TABLE audit_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER,
    action_type TEXT NOT NULL,
    resource_type TEXT NOT NULL,
    resource_id TEXT,
    details TEXT,
    ip_address TEXT,
    user_agent TEXT,
    result TEXT,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
    session_id TEXT
);

-- 脚本执行日志表
CREATE TABLE script_execution_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    task_id TEXT NOT NULL,
    script_id INTEGER,
    host_id INTEGER,
    user_id INTEGER,
    start_time DATETIME NOT NULL,
    end_time DATETIME,
    exit_code INTEGER,
    stdout TEXT,
    stderr TEXT,
    command TEXT NOT NULL,
    parameters TEXT,
    status TEXT NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(script_id) REFERENCES scripts(id),
    FOREIGN KEY(host_id) REFERENCES hosts(id)
);
```
